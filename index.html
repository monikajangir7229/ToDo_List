let input = document.querySelector("input"); let task = document.querySelector("#addBtn"); let delAll = document.querySelector("#clearBtn"); let ul = document.querySelector("ul"); let totalTask = document.querySelector("#taskCount"); task.addEventListener("click", function () { if (input.value.trim() === "") { return; } let list = document.createElement("li"); list.classList.add("task"); let span = document.createElement("span"); span.innerText = input.value; let actions = document.createElement("div"); actions.classList.add("actions"); let doneBtn = document.createElement("button"); doneBtn.innerText = "✓"; doneBtn.classList.add("done-btn"); let delBtn = document.createElement("button"); delBtn.innerText = "🗑"; delBtn.classList.add("delete-btn"); actions.appendChild(doneBtn); actions.appendChild(delBtn); list.appendChild(span); list.appendChild(actions); ul.insertAdjacentElement("afterbegin", list); input.value = ""; delBtn.addEventListener("click", function () { list.remove(); totalTask.innerText = ul.children.length; }); doneBtn.addEventListener("click", function () { list.classList.toggle("completed"); }); delAll.addEventListener("click", function () { ul.innerHTML = ""; totalTask.innerText = "0"; }); totalTask.innerText = ul.children.length; });
